# =============================================================================
# OCR sidecar -- REAL-backend (OCR_MODE=ppocr) dependencies.
#
# These install into the sidecar's OWN venv (sidecars/ocr/.venv) and NEVER into
# the shared data-processing venv. That quarantine is the whole reason WS-C is a
# separate deployable (D-06): the PP-OCR stack drags onnxruntime, opencv, shapely
# and pyclipper, and the paddleocr path additionally pins `numpy<2.4`, which
# conflicts with DP's installed numpy 2.5.1.
#
#   OCR_MODE=mock needs NONE of this -- it is Python-stdlib only. Only the real
#   `ppocr` backend imports these, lazily, at engine construction.
#
# Set up the venv:
#   python3 -m venv sidecars/ocr/.venv
#   sidecars/ocr/.venv/bin/pip install -r sidecars/ocr/requirements.txt
#
# Pins reflect the validated CPU stack (Python 3.12). onnxruntime is pinned
# because its version is reported at /health and is version-relevant (ORT is not
# guaranteed bit-exact across releases/providers).
# =============================================================================

# PP-OCR det+rec ONNX on CPU via onnxruntime. Bundles the ch_PP-OCRv4 det/rec
# ONNX models; numpy-2.x compatible (unlike paddleocr). To run PP-OCRv5/v6 (or
# any other det+rec pair), point OCR_DET_MODEL / OCR_REC_MODEL at the .onnx files
# and, if needed, OCR_REC_DICT at the rec char dictionary -- /health re-hashes
# whatever loads, so the model is a file swap (D-06).
rapidocr-onnxruntime==1.4.4

# CPU ONNX Runtime. Pinned: /health reports ort_version and the EP.
onnxruntime==1.27.0

# JPEG decode for the /ocr request payload.
pillow==12.3.0

# Pulled in transitively by rapidocr-onnxruntime, pinned here for reproducibility:
numpy==2.5.1
opencv-python==5.0.0.93
shapely==2.1.2
pyclipper==1.4.0
PyYAML==6.0.3
